% Shared preamble for every resume in this repo.
% Facts live in the other shared/*.tex files; each resume only picks and orders them.
% Each resume file defines \ifonepage and \ifai (with \newif) BEFORE inputting this file:
%   \onepagetrue  -> 1-page layout: smaller margins/font/spacing, fewer bullets
%   \aitrue       -> AI work leads the summary, skills and Oracle bullets (AI versions)
\ifonepage
  \documentclass[10pt]{article}
\else
  \documentclass[10.5pt]{article}
\fi

\usepackage[T1]{fontenc}
\usepackage[utf8]{inputenc}
% Latin Modern: scalable Type 1 version of Computer Modern (the same design the
% Overleaf builds show). Without it, pdfLaTeX + T1 fontenc falls back to Type 3
% bitmap fonts, which render fuzzy in PDF viewers. Ships with
% texlive-latex-recommended, so no CI package changes are needed.
\usepackage{lmodern}
\ifonepage
  \usepackage[a4paper,margin=0.5in]{geometry}
\else
  \usepackage[a4paper,margin=0.65in]{geometry}
\fi

\pagestyle{empty}
\setlength{\parindent}{0pt}

% Section heading: bold caps with a rule underneath (plain LaTeX, no extra packages).
% A \nopagebreak sits BEFORE every internal \vspace: a break is only legal at a
% penalty, and a penalty 10000 makes the next glue non-breakable. This glues the
% heading, its rule and the entry that follows into one unbreakable block, so a
% heading can never sit alone at a page bottom; TeX moves the whole block to the
% next page instead. The entry macros do the same for the header line and its
% first bullet (with \@beginparpenalty in tightitems). No needspace package:
% forced pre-measured breaks caused thin trailing pages.
\ifonepage
  \newcommand{\rsection}[1]{\vspace{5pt}{\bfseries\MakeUppercase{#1}}\par\nopagebreak\vspace{2pt}\nopagebreak\hrule height 0.6pt\nopagebreak\vspace{3pt}\nopagebreak}
\else
  \newcommand{\rsection}[1]{\vspace{7pt}{\bfseries\MakeUppercase{#1}}\par\nopagebreak\vspace{2pt}\nopagebreak\hrule height 0.6pt\nopagebreak\vspace{4pt}\nopagebreak}
\fi
% Tight bullet list without enumitem
% \@beginparpenalty=10000 stops a page break between an entry header and its first
% bullet, so a job/project title never sits alone at the bottom of a page.
\makeatletter
\ifonepage
  \newenvironment{tightitems}{\@beginparpenalty=10000 \begin{itemize}\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}\setlength{\topsep}{0pt}\setlength{\leftmargin}{1.2em}}{\end{itemize}}
\else
  \newenvironment{tightitems}{\@beginparpenalty=10000 \begin{itemize}\setlength{\itemsep}{1pt}\setlength{\parskip}{0pt}\setlength{\topsep}{1pt}\setlength{\leftmargin}{1.2em}}{\end{itemize}}
\fi
\makeatother

% Job/entry header: bold title line, details after a pipe
\ifonepage
  \newcommand{\entry}[4]{%
    \vspace{2pt}\textbf{#1, #2}\hspace{4pt}{\textbar}\hspace{4pt}#3\hspace{4pt}{\textbar}\hspace{4pt}#4\par\nopagebreak\vspace{1pt}\nopagebreak}
  \newcommand{\entrynodate}[3]{%
    \vspace{2pt}\textbf{#1, #2}\hspace{4pt}{\textbar}\hspace{4pt}#3\par\nopagebreak\vspace{1pt}\nopagebreak}
\else
  \newcommand{\entry}[4]{%
    \vspace{3pt}\textbf{#1, #2}\hspace{4pt}{\textbar}\hspace{4pt}#3\hspace{4pt}{\textbar}\hspace{4pt}#4\par\nopagebreak\vspace{1pt}\nopagebreak}
  \newcommand{\entrynodate}[3]{%
    \vspace{3pt}\textbf{#1, #2}\hspace{4pt}{\textbar}\hspace{4pt}#3\par\nopagebreak\vspace{1pt}\nopagebreak}
\fi

% Years of experience, computed at compile time from the Sep 2012 start date.
% Prints the whole number; use as \YOE+ (e.g. "14+" until Aug 2027, "15+" from Sep 2027).
\newcommand{\YOE}{\the\numexpr\year-2012\ifnum\month<9 -1\fi\relax}
